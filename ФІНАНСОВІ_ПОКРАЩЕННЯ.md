### Проблема 3: "Скільки буде коштувати проект?"

### Проблема 4: "Де я переплачую?"
**Рішення:** Аналіз ефективності та рекомендації

```typescript
interface PayrollEntry {
  userId: string;
  userName: string;
  level: string;
  hourlyRate: number;

  // Деталі
  regularHours: number;        // Звичайні години
  overtimeHours: number;        // Понаднормові (якщо >160/місяць)
  businessTripHours: number;    // Відрядження
  processEarnings: number;      // За процеси
  bonuses: number;              // Премії
  deductions: number;           // Утримання (аванси, штрафи)

  // Підсумки
  grossSalary: number;          // Нарахована
  tax: number;                  // Податок (18%)
  militaryTax: number;          // Військовий збір (1.5%)
  netSalary: number;            // До виплати

  // Статус
  advancePaid: number;          // Виплачений аванс
  remainingToPay: number;       // Залишок до виплати
  paymentStatus: 'pending' | 'partially_paid' | 'paid';
}

const calculatePayroll = (period: { start: string; end: string }) => {
  const entries: PayrollEntry[] = [];

  users.forEach(user => {
    // Години
    const userHours = hours.filter(h =>
      h.userId === user.id &&
      h.date >= period.start &&
      h.date <= period.end
    );

    const regularHours = userHours
      .filter(h => !h.isBusinessTrip)
      .reduce((sum, h) => sum + h.hours, 0);

    const businessTripHours = userHours
      .filter(h => h.isBusinessTrip)
      .reduce((sum, h) => sum + h.hours, 0);

    // Понаднормові (якщо >160 годин/місяць)
    const overtimeHours = Math.max(0, regularHours - 160);
    const normalHours = regularHours - overtimeHours;

    // Процеси
    const userProcesses = processes.filter(p =>
      p.userId === user.id &&
      p.date >= period.start &&
      p.date <= period.end
    );
    const processEarnings = userProcesses.reduce((sum, p) => sum + p.salary, 0);

    // Розрахунки
    const regularPay = normalHours * user.hourlyRate;
    const overtimePay = overtimeHours * user.hourlyRate * 1.5; // +50% за понаднормові
    const businessTripPay = businessTripHours * user.hourlyRate * 1.2; // +20% за відрядження

    const grossSalary = regularPay + overtimePay + businessTripPay + processEarnings;
    const tax = grossSalary * 0.18; // ПДФО
    const militaryTax = grossSalary * 0.015; // Військовий збір
    const netSalary = grossSalary - tax - militaryTax;

    entries.push({
      userId: user.id,
      userName: user.name,
      level: user.level,
      hourlyRate: user.hourlyRate,
      regularHours: normalHours,
      overtimeHours,
      businessTripHours,
      processEarnings,
      bonuses: 0,
      deductions: 0,
      grossSalary,
      tax,
      militaryTax,
      netSalary,
      advancePaid: 0,
      remainingToPay: netSalary,
      paymentStatus: 'pending'
    });
  });

  return entries;
};
```

## 2. 🔍 Перехресна Перевірка та Виявлення Аномалій

### Автоматична Перевірка:

```typescript
interface AnomalyCheck {
  userId: string;
  type: 'suspicious_hours' | 'impossible_timeline' | 'unusual_pattern' | 'duplicate_entry';
  severity: 'low' | 'medium' | 'high';
  description: string;
  details: any;
  date: string;
}

const detectAnomalies = () => {
  const anomalies: AnomalyCheck[] = [];

  // 1. Перевірка на занадто багато годин на день
  const hoursByDay = groupBy(hours, h => `${h.userId}-${h.date}`);
  Object.entries(hoursByDay).forEach(([key, dayHours]) => {
    const total = dayHours.reduce((sum, h) => sum + h.hours, 0);
    if (total > 12) {
      anomalies.push({
        userId: dayHours[0].userId,
        type: 'suspicious_hours',
        severity: 'high',
        description: `${total} годин за день - можливо помилка`,
        details: { date: dayHours[0].date, hours: total },
        date: dayHours[0].date
      });
    }
  });

  // 2. Робота на двох об'єктах одночасно
  const sameTimeWork = hours.filter((h, i) => {
    return hours.some((h2, j) =>
      i !== j &&
      h.userId === h2.userId &&
      h.date === h2.date &&
      h.object !== h2.object
    );
  });

  sameTimeWork.forEach(h => {
    anomalies.push({
      userId: h.userId,
      type: 'impossible_timeline',
      severity: 'medium',
      description: 'Працював на 2+ об\'єктах в один день',
      details: { date: h.date, objects: [...new Set(sameTimeWork.map(x => x.object))] },
      date: h.date
    });
  });

  // 3. Незвичайні шаблони (працює тільки вихідні)
  const userWorkDays = groupBy(hours, 'userId');
  Object.entries(userWorkDays).forEach(([userId, userHours]) => {
    const weekendWork = userHours.filter(h => {
      const day = new Date(h.date).getDay();
      return day === 0 || day === 6; // Неділя або Субота
    });

    const weekendRatio = weekendWork.length / userHours.length;
    if (weekendRatio > 0.5 && userHours.length > 10) {
      anomalies.push({
        userId,
        type: 'unusual_pattern',
        severity: 'low',
        description: `${(weekendRatio * 100).toFixed(0)}% роботи у вихідні`,
        details: { weekendCount: weekendWork.length, totalDays: userHours.length },
        date: new Date().toISOString()
      });
    }
  });

  // 4. Дублікати (однакові записи)
  const duplicates = hours.filter((h, i) =>
    hours.some((h2, j) =>
      i < j &&
      h.userId === h2.userId &&
      h.date === h2.date &&
      h.object === h2.object &&
      h.hours === h2.hours
    )
  );

  duplicates.forEach(h => {
    anomalies.push({
      userId: h.userId,
      type: 'duplicate_entry',
      severity: 'high',
      description: 'Можливий дублікат запису',
      details: { date: h.date, object: h.object, hours: h.hours },
      date: h.date
    });
  });

  return anomalies;
};
```

```typescript
  netProfit: number;            // Чистий прибуток (після податків)

  // Ефективність
  hoursPlanned: number;
  hoursActual: number;
  efficiency: number;           // % виконання

  // Проблеми
  issues: string[];
  recommendations: string[];
}

const analyzeProfitability = (objectName: string): ProfitabilityAnalysis => {
  const obj = objects.find(o => o.name === objectName);
  const stats = objectStats[objectName];

  // Розрахунки
  const revenue = obj.plannedBudget || 0;
  const laborCost = stats.earnings;
  const materialsCost = stats.materialsCost;
  const equipmentCost = 0; // TODO: додати в майбутньому
  const overheadCost = laborCost * 0.1; // 10% накладних

  const totalCost = laborCost + materialsCost + equipmentCost + overheadCost;
  const grossProfit = revenue - totalCost;
  const grossMargin = revenue > 0 ? (grossProfit / revenue) * 100 : 0;

  const tax = grossProfit * 0.18; // Спрощений приклад
  const netProfit = grossProfit - tax;

  const hoursPlanned = obj.maxHours || 0;
  const hoursActual = stats.hoursCount;
  const efficiency = hoursPlanned > 0 ? (hoursActual / hoursPlanned) * 100 : 0;

  // Виявлення проблем
  const issues = [];
  const recommendations = [];

  if (grossMargin < 15) {
    issues.push('❌ Низька маржа (<15%)');
    recommendations.push('Підвищити ціну або зменшити витрати');
  }

  if (efficiency > 110) {
    issues.push('⚠️ Перевитрата годин (+10%)');
    recommendations.push('Покращити планування або швидкість роботи');
  }

  if (materialsCost / totalCost > 0.4) {
    issues.push('📦 Високі витрати на матеріали (>40%)');
    recommendations.push('Шукати дешевших постачальників');
  }

  const avgHourlyRate = laborCost / hoursActual;
  if (avgHourlyRate > 250) {
    issues.push('💰 Висока середня ставка');
    recommendations.push('Розподілити завдання ефективніше (юніори на прості роботи)');
  }

  return {
    objectName,
    revenue,
    laborCost,
    materialsCost,
    equipmentCost,
    overheadCost,
    totalCost,
    grossProfit,
    grossMargin,
    netProfit,
    hoursPlanned,
    hoursActual,
    efficiency,
    issues,
    recommendations
  };
};
```
